## 3. 48h 자동 루프 스케줄

```yaml
# cron_48h.yaml
loop:
  - hour: 0
    jobs:
      - P1-A: config_diff
      - P1-B: cache_clear_champion
      - P1-C: cache_clear_candidate
    gate: "P1-A.weight_diff > 0"
    on_fail: "HALT + alert"

  - hour: 6
    jobs:
      - P2-A: fold_breakdown
      - P2-B: regime_check
    gate: "P2-A.fold_std < 0.5"
    on_fail: "FLAG overfitting"

  - hour: 12
    jobs:
      - P3-A: healthcare_weight_test
      - P4-A: cost_10bp
    gate: "P3-A.mdd > -0.10 AND P4-A.delta < 0.15"
    on_fail: "CONTINUE with warning"

  - hour: 24
    jobs:
      - P3-B: vol_target_test
      - P4-B: cost_20bp
    gate: "tuning_mean >= 1.5"
    on_fail: "ROLLBACK to champion"

  - hour: 36
    jobs:
      - P5-A: hhi_check
      - P5-C: healthcare_expansion
    gate: "HHI < 0.15"
    on_fail: "REBALANCE weights"

  - hour: 48
    jobs:
      - HANDOFF: generate_next_champion
```
