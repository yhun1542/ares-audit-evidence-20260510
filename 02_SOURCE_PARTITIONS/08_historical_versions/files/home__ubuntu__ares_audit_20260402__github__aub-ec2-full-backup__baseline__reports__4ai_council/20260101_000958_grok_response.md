def get_dynamic_top_k(self, regime: str, icir: float) -> int:
    """ICIR + regime adaptive top_k (CLI override via --top_k as base)."""
    base_k = 10  # Default
    vol_scalars = {
        "calm_bull": 1.5, "volatile_bull": 1.0, "tightening": 0.7,
        "bear_market": 0.6, "crisis": 0.5, "slow_risk": 0.8
    }
    scalar = vol_scalars.get(regime, 1.0)
    adj_k = max(5, min(15, int(base_k * scalar * max(0.5, icir))))  # Clamp 5-15; ICIR floor
    if icir < 0.02:  # Fold16-like
        adj_k = min(adj_k, 7)
    return adj_k
