python3 -c "
import json
d = json.load(open('baseline/champion_v5_latest.json'))
returns = d.get('daily_returns', [])
if returns:
    cummax = 0
    mdd = 0
    for r in returns:
        cummax = max(cummax, cummax*(1+r))
        dd = (cummax - cummax*(1+r))/cummax if cummax else 0
        mdd = min(mdd, -dd)
    print(f'worst_mdd: {mdd:.4f}')
else:
    print('No daily_returns found')
"
