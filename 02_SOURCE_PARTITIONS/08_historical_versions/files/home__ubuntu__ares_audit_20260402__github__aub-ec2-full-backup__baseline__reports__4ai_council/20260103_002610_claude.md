## 2. 48H 자동 루프 스펙

```yaml
# experiment_system/config/48h_loop_20260103.yaml

experiments:
  - id: EXP-1A
    module: gate/mdd_gate.py
    param: {worst_mdd_threshold: -0.085}
    gate: "lockbox_mean >= 1.5 AND worst_mdd >= -0.09"
    observe: [lockbox_mean, worst_mdd, mdd_std]

  - id: EXP-2B
    module: position/sector_cap.py
    param: {tech_max: 0.45}
    gate: "lockbox_mean >= prev*0.95 AND worst_mdd_improvement >= 0.10"
    observe: [tech_exposure, worst_mdd, lockbox_mean]

  - id: EXP-3A
    module: regime/vix_filter.py
    param: {vix_threshold: 25}
    gate: "0.5 <= tuning_lockbox_ratio <= 0.8"
    observe: [tuning_mean, lockbox_mean, regime_switch_count]

  - id: EXP-4A
    module: allocation/sector_floor.py
    param: {healthcare_min: 0.15}
    gate: "lockbox_mean >= 1.6 AND sector_corr < 0.55"
    observe: [healthcare_weight, sector_corr, lockbox_mean]

  - id: EXP-5B
    module: cost/cost_model.py
    param: {slippage_bp: 8}
    gate: "net_gross_ratio >= 0.92"
    observe: [gross_return, net_return, turnover]

sequence: parallel  # 독립 실험, 동시 실행
promote_rule: "ALL gates PASS → combine into candidate_v6"
fallback: "FAIL 2+ → revert to champion_v5, flag for manual review"
```
