### 아이디어 1: Momentum Reversal Detector (MRD)
| 항목 | 내용 |
|------|------|
| **예상 효과** | OOS Sharpe +0.3~0.5 |
| **리스크** | 과민 반응 시 불필요한 헤지 비용 |
| **구현 난이도** | 중간 |
| **우선순위** | ⭐⭐⭐⭐⭐ (1순위) |

**원리**: 모멘텀 팩터의 급격한 약화를 감지하여 사전 방어

```python
# momentum_reversal_detector.py
from dataclasses import dataclass
from typing import Optional, Tuple
import numpy as np

@dataclass
class MRDConfig:
    """Momentum Reversal Detector 설정"""
    lookback_short: int = 5
    lookback_long: int = 20
    reversal_threshold: float = -0.15  # 15% 이상 하락 시 경고
    acceleration_threshold: float = -0.05  # 가속도 임계값
    position_scale_on_reversal: float = 0.4
    hedge_boost_on_reversal: float = 0.25

class MomentumReversalDetector:
    """모멘텀 반전 감지기 - Fold 16 대응"""
    
    def __init__(self, config: Optional[MRDConfig] = None):
        self.config = config or MRDConfig()
        self._momentum_history: list[float] = []
        
    def update(self, momentum_score: float) -> None:
        """모멘텀 스코어 업데이트"""
        self._momentum_history.append(momentum_score)
        # 메모리 관리: 최근 60일만 유지
        if len(self._momentum_history) > 60:
            self._momentum_history.pop(0)
    
    def detect(self) -> Tuple[bool, dict]:
        """
        모멘텀 반전 감지
        
        Returns:
            Tuple[bool, dict]: (반전 감지 여부, 조정 파라미터)
        """
        if len(self._momentum_history) < self.config.lookback_long:
            return False, {}
        
        hist = np.array(self._momentum_history)
        
        # 단기/장기 모멘텀 변화율
        short_change = (hist[-1] - hist[-self.config.lookback_short]) / (abs(hist[-self.config.lookback_short]) + 1e-8)
        long_mean = np.mean(hist[-self.config.lookback_long:])
        
        # 가속도 계산 (2차 미분)
        if len(hist) >= 10:
            recent_changes = np.diff(hist[-10:])
            acceleration = np.mean(recent_changes[-3:]) - np.mean(recent_changes[:3])
        else:
            acceleration = 0.0
        
        # 반전 조건
        is_reversal = (
            short_change < self.config.reversal_threshold or 
            (short_change < -0.08 and acceleration < self.config.acceleration_threshold)
        )
        
        if is_reversal:
            return True, {
                "position_scale": self.config.position_scale_on_reversal,
                "hedge_weight_boost": self.config.hedge_boost_on_reversal,
                "enter_days_override": 0,  # 즉시 대응
                "exit_days_override": 7,
                "signal_strength": abs(short_change),
            }
        
        return False, {}
    
    def get_diagnostics(self) -> dict:
        """진단 정보 반환"""
        if len(self._momentum_history) < 5:
            return {"status": "insufficient_data"}
        
        hist = np.array(self._momentum_history)
        return {
            "current_momentum": hist[-1],
            "momentum_5d_change": (hist[-1] - hist[-5]) if len(hist) >= 5 else None,
            "momentum_20d_mean": np.mean(hist[-20:]) if len(hist) >= 20 else None,
            "history_length": len(self._momentum_history),
        }


# 테스트 코드
def test_momentum_reversal_detector():
    """MRD 단위 테스트"""
    detector = MomentumReversalDetector()
    
    # 정상 상승 구간
    for i in range(25):
        detector.update(0.5 + i * 0.01)
    is_reversal, params = detector.detect()
    assert not is_reversal, "정상 상승 시 반전 감지되면 안됨"
    
    # 급격한 하락 구간 (Fold 16 시뮬레이션)
    for i in range(5):
        detector.update(0.7 - i * 0.08)  # 급락
    is_reversal, params = detector.detect()
    assert is_reversal, "급락 시 반전 감지되어야 함"
    assert params["position_scale"] < 0.5, "포지션 축소 필요"
    
    print("✅ MomentumReversalDetector 테스트 통과")
    return True

if __name__ == "__main__":
    test_momentum_reversal_detector()
```

### 아이디어 2: Adaptive Regime Smoothing (ARS)
| 항목 | 내용 |
|------|------|
| **예상 효과** | OOS Sharpe +0.2~0.3 |
| **리스크** | 레짐 전환 지연 가능 |
| **구현 난이도** | 낮음 |
| **우선순위** | ⭐⭐⭐⭐⭐ (1순위) |

**원리**: 레짐 전환 시 급격한 파라미터 변화 방지, 점진적 전환

```python
# adaptive_regime_smoothing.py
from dataclasses import dataclass, field
from typing import Dict, Optional
from collections import deque

@dataclass
class ARSConfig:
    """Adaptive Regime Smoothing 설정"""
    smoothing_window: int = 3  # 레짐 확정까지 필요한 연속 일수
    transition_speed: float = 0.3  # 파라미터 전환 속도 (0.1=느림, 0.5=빠름)
    min_confidence: float = 0.6  # 레짐 확정 최소 신뢰도

@dataclass
class RegimeState:
    """레짐 상태 추적"""
    current_regime: str = "calm_bull"
    pending_regime: Optional[str] = None
    pending_count: int = 0
    transition_progress: float = 1.0  # 0=이전 레짐, 1=현재 레짐
    regime_history: deque = field(default_factory=lambda: deque(maxlen=20))

class AdaptiveRegimeSmoothing:
    """레짐 전환 스무딩 - 급격한 파라미터 변화 방지"""
    
    # 기존 MacroAlphaAddon의 REGIME_PARAMS 참조
    REGIME_PARAMS = {
        "calm_bull": {"position_scale": 1.0, "hedge_weight_on": 0.08},
        "volatile_bull": {"position_scale": 0.85, "hedge_weight_on": 0.15},
        "tightening": {"position_scale": 0.7, "hedge_weight_on": 0.20},
        "bear_market": {"position_scale": 0.5, "hedge_weight_on": 0.30},
        "crisis": {"position_scale": 0.3, "hedge_weight_on": 0.50},
        "slow_risk": {"position_scale": 0.6, "hedge_weight_on": 0.25},
    }
    
    # 레짐 전환 우선순위 (위험 레짐으로는 빠르게, 안전 레짐으로는 느리게)
    REGIME_RISK_LEVEL = {
        "calm_bull": 0,
        "volatile_bull": 1,
        "slow_risk": 2,
        "tightening": 3,
        "bear_market": 4,
        "crisis": 5,
    }
    
    def __init__(self, config: Optional[ARSConfig] = None):
        self.config = config or ARSConfig()
        self.state = RegimeState()
        self._prev_params: Dict[str, float] = {}
    
    def update_regime(self, detected_regime: str) -> Dict[str, float]:
        """
        레짐 업데이트 및 스무딩된 파라미터 반환
        
        Args:
            detected_regime: 감지된 레짐
            
        Returns:
            Dict: 스무딩된 파라미터
        """
        self.state.regime_history.append(detected_regime)
        
        # 레짐 변경 감지
        if detected_regime != self.state.current_regime:
            if detected_regime == self.state.pending_regime:
                self.state.pending_count += 1
            else:
                self.state.pending_regime = detected_regime
                self.state.pending_count = 1
            
            # 위험 레짐으로 전환 시 빠른 전환
            current_risk = self.REGIME_RISK_LEVEL.get(self.state.current_regime, 0)
            pending_risk = self.REGIME_RISK_LEVEL.get(detected_regime, 0)
            
            required_count = self.config.smoothing_window
            if pending_risk > current_risk:
                required_count = max(1, self.config.smoothing_window - (pending_risk - current_risk))
            
            # 레짐 확정
            if self.state.pending_count >= required_count:
                self._prev_params = self.REGIME_PARAMS.get(
                    self.state.current_regime, 
                    self.REGIME_PARAMS["calm_bull"]
                ).copy()
                self.state.current_regime = detected_regime
                self.state.pending_regime = None
                self.state.pending_count = 0
                self.state.transition_progress = 0.0
        else:
            self.state.pending_regime = None
            self.state.pending_count = 0
        
        # 점진적 전환
        if self.state.transition_progress < 1.0:
            self.state.transition_progress = min(
                1.0, 
                self.state.transition_progress + self.config.transition_speed
            )
        
        return self._get_smoothed_params()
    
    def _get_smoothed_params(self) -> Dict[str, float]:
        """스무딩된 파라미터 계산"""
        current_params = self.REGIME_PARAMS.get(
            self.state.current_regime, 
            self.REGIME_PARAMS["calm_bull"]
        )
        
        if not self._prev_params or self.state.transition_progress >= 1.0:
            return current_params.copy()
        
        # 선형 보간
        smoothed = {}
        for key in current_params:
            prev_val = self._prev_params.get(key, current_params[key])
            curr_val = current_params[key]
            smoothed[key] = prev_val + (curr_val - prev_val) * self.state.transition_progress
        
        return smoothed
    
    def get_regime_confidence(self) -> float:
        """현재 레짐 신뢰도"""
        if len(self.state.regime_history) < 5:
            return 0.5
        
        recent = list(self.state.regime_history)[-5:]
        same_count = sum(1 for r in recent if r == self.state.current_regime)
        return same_count / len(recent)


def test_adaptive_regime_smoothing():
    """ARS 단위 테스트"""
    ars = AdaptiveRegimeSmoothing()
    
    # 정상 상태
    params = ars.update_regime("calm_bull")
    assert params["position_scale"] == 1.0
    
    # 급격한 레짐 변화 (crisis로)
    params = ars.update_regime("crisis")
    assert params["position_scale"] > 0.3, "첫 날은 스무딩되어야 함"
    
    # 연속 crisis
    for _ in range(3):
        params = ars.update_regime("crisis")
    assert params["position_scale"] <= 0.5, "레짐 확정 후 전환 진행"
    
    print("✅ AdaptiveRegimeSmoothing 테스트 통과")
    return True

if __name__ == "__main__":
    test_adaptive_regime_smoothing()
```

### 아이디어 3: Fed Signal Anticipator (FSA)
| 항목 | 내용 |
|------|------|
| **예상 효과** | OOS Sharpe +0.4~0.6 (Fold 16 특화) |
| **리스크** | Fed 이벤트 외 기간에는 효과 미미 |
| **구현 난이도** | 중간 |
| **우선순위** | ⭐⭐⭐⭐ (2순위) |

**원리**: Fed 이벤트 전후 특수 레짐 적용 (Fold 16 = 테이퍼링)

```python
# fed_signal_anticipator.py
from dataclasses import dataclass
from datetime import datetime, timedelta
from typing import Optional, Dict, List
import numpy as np

@dataclass
class FSAConfig:
    """Fed Signal Anticipator 설정"""
    pre_event_days: int = 5  # 이벤트 전 방어 시작
    post_event_days: int = 10  # 이벤트 후 관찰 기간
    position_scale_pre: float = 0.6
    position_scale_post: float = 0.5
    hedge_boost_pre: float = 0.15
    hedge_boost_post: float = 0.25
    volatility_threshold: float = 0.02  # 일일 변동성 임계값

@dataclass
class FedEvent:
    """Fed 이벤트 정의"""
    date: datetime
    event_type: str  # "fomc", "tapering", "rate_decision", "speech"
    importance: int  # 1-5
    expected_direction: str  # "hawkish", "dovish", "neutral"

class FedSignalAnticipator:
    """Fed 이벤트 선제 대응"""
    
    # 2021 주요 Fed 이벤트 (Fold 16 대응)
    HISTORICAL_EVENTS_2021 = [
        FedEvent(datetime(2021, 6, 16), "fomc", 4, "hawkish"),
        FedEvent(datetime(2021, 7, 28), "fomc", 3, "neutral"),
        FedEvent(datetime(2021, 8, 27), "speech", 5, "hawkish"),  # Jackson Hole
        FedEvent(datetime(2021, 9, 22), "fomc", 5, "hawkish"),    # Tapering 시사
        FedEvent(datetime(2021, 11, 3), "tapering", 5, "hawkish"), # Tapering 발표
        FedEvent(datetime(2021, 12, 15), "fomc", 5, "hawkish"),   # 가속화
    ]
    
    def __init__(self, config: Optional[FSAConfig] = None):
        self.config = config or FSAConfig()
        self._events: List[FedEvent] = self.HISTORICAL_EVENTS_2021.copy()
        self._volatility_history: List[float] = []
    
    def add_event(self, event: FedEvent) -> None:
        """이벤트 추가"""
        self._events.append(event)
        self._events.sort(key=lambda x: x.date)
    
    def update_volatility(self, daily_return: float) -> None:
        """일일 변동성 업데이트"""
        self._volatility_history.append(abs(daily_return))
        if len(self._volatility_history) > 60:
            self._volatility_history.pop(0)
    
    def get_event_adjustment(self, current_date: datetime) -> Dict[str, float]:
        """
        현재 날짜 기준 이벤트 조정 파라미터
        
        Returns:
            Dict: 조정 파라미터 (비어있으면 조정 없음)
        """
        adjustments = {}
        
        for event in self._events:
            days_to_event = (event.date - current_date).days
            days_since_event = (current_date - event.date).days
            
            # 이벤트 전 방어
            if 0 < days_to_event <= self.config.pre_event_days:
                intensity = 1 - (days_to_event / self.config.pre_event_days)
                intensity *= (event.importance / 5)
                
                adjustments["position_scale"] = min(
                    adjustments.get("position_scale", 1.0),
                    1.0 - (1.0 - self.config.position_scale_pre) * intensity
                )
                adjustments["hedge_boost"] = max(
                    adjustments.get("hedge_boost", 0),
                    self.config.hedge_boost_pre * intensity
                )
                adjustments["event_phase"] = "pre"
                adjustments["days_to_event"] = days_to_event
            
            # 이벤트 후 관찰
            elif 0 <= days_since_event <= self.config.post_event_days:
                intensity = 1 - (days_since_event / self.config.post_event_days)
                intensity *= (event.importance / 5)
                
                # Hawkish 이벤트 후 더 보수적
                if event.expected_direction == "hawkish":
                    intensity *= 1.3
                
                adjustments["position_scale"] = min(
                    adjustments.get("position_scale", 1.0),
                    1.0 - (1.0 - self.config.position_scale_post) * intensity
                )
                adjustments["hedge_boost"] = max(
                    adjustments.get("hedge_boost", 0),
                    self.config.hedge_boost_post * intensity
                )
                adjustments["event_phase"] = "post"
                adjustments["days_since_event"] = days_since_event
        
        # 변동성 기반 추가 조정
        if len(self._volatility_history) >= 5:
            recent_vol = np.mean(self._volatility_history[-5:])
            if recent_vol > self.config.volatility_threshold:
                vol_multiplier = min(2.0, recent_vol / self.config.volatility_threshold)
                if "position_scale" in adjustments:
                    adjustments["position_scale"] *= (1 / vol_multiplier)
                if "hedge_boost" in adjustments:
                    adjustments["hedge_boost"] *= vol_multiplier
        
        return adjustments
    
    def is_event_window(self, current_date: datetime) -> bool:
        """현재 이벤트 윈도우 내인지 확인"""
        for event in self._events:
            days_diff = abs((current_date - event.date).days)
            window = self.config.pre_event_days + self.config.post_event_days
            if days_diff <= window:
                return True
        return False


def test_fed_signal_anticipator():
    """FSA 단위 테스트"""
    fsa = FedSignalAnticipator()
    
    # 이벤트 5일 전
    test_date = datetime(2021, 9, 17)  # FOMC 5일 전
    adj = fsa.get_event_adjustment(test_date)
    assert "position_scale" in adj, "이벤트 전 조정 필요"
    assert adj["position_scale"] < 1.0, "포지션 축소 필요"
    
    # 이벤트 당일
    test_date = datetime(2021, 9, 22)
    adj = fsa.get_event_adjustment(test_date)
    assert adj.get("event_phase") == "post"
    
    # 평상시
    test_date = datetime(2021, 5, 1)
    adj = fsa.get_event_adjustment(test_date)
    assert len(adj) == 0, "이벤트 없는 날 조정 없어야 함"
    
    print("✅ FedSignalAnticipator 테스트 통과")
    return True

if __name__ == "__main__":
    test_fed_signal_anticipator()
```

### 아이디어 4: Dynamic ICIR Weighting
| 항목 | 내용 |
|------|------|
| **예상 효과** | OOS Sharpe +0.15~0.25 |
| **리스크** | ICIR 계산 비용 증가 |
| **구현 난이도** | 낮음 |
| **우선순위** | ⭐⭐⭐⭐ (2순위) |

**원리**: 팩터별 최근 ICIR 기반 동적 가중치 조정

```python
# dynamic_icir_weighting.py
from dataclasses import dataclass
from typing import Dict, List, Optional
import numpy as np
from collections import defaultdict

@dataclass
class DICIRConfig:
    """Dynamic ICIR Weighting 설정"""
    lookback_period: int = 20
    min_weight: float = 0.05
    max_weight: float = 0.5
    decay_factor: float = 0.9  # 최근 데이터 가중치
    negative_icir_penalty: float = 0.5  # 음수 ICIR 페널티

class DynamicICIRWeighting:
    """동적 ICIR 기반 팩터 가중치"""
    
    def __init__(self, config: Optional[DICIRConfig] = None):
        self.config = config or DICIRConfig()
        self._factor_returns: Dict[str, List[float]] = defaultdict(list)
        self._factor_ics: Dict[str, List[float]] = defaultdict(list)
    
    def update(self, factor_name: str, ic_value: float, factor_return: float) -> None:
        """팩터 데이터 업데이트"""
        self._factor_ics[factor_name].append(ic_value)
        self._factor_returns[factor_name].append(factor_return)
        
        # 메모리 관리
        max_history = self.config.lookback_period * 3
        if len(self._factor_ics[factor_name]) > max_history:
            self._factor_ics[factor_name] = self._factor_ics[factor_name][-max_history:]
            self._factor_returns[factor_name] = self._factor_returns[factor_name][-max_history:]
    
    def get_weights(self) -> Dict[str, float]:
        """현재 최적 가중치 계산"""
        icir_scores = {}
        
        for factor_name in self._factor_ics:
            ics = self._factor_ics[factor_name][-self.config.lookback_period:]
            
            if len(ics) < 5:
                icir_scores[factor_name] = 0.0
                continue
            
            # Exponential weighting for recency
            weights = np.array([self.config.decay_factor ** i for i in range(len(ics)-1, -1, -1)])
            weights /= weights.sum()
            
            weighted_mean = np.sum(np.array(ics) * weights)
            weighted_std = np.sqrt(np.sum(weights * (np.array(ics) - weighted_mean) ** 2))
            
            if weighted_std > 1e-8:
                icir = weighted_mean / weighted_std
            else:
                icir = 0.0
            
            # 음수 ICIR 페널티
            if icir < 0:
                icir *= self.config.negative_icir_penalty
            
            icir_scores[factor_name] = icir
        
        # Softmax 정규화
        if not icir_scores:
            return {}
        
        scores = np.array(list(icir_scores.values()))
        exp_scores = np.exp(scores - np.max(scores))  # numerical stability
        softmax_weights = exp_scores / exp_scores.sum()
        
        # min/max 제약 적용
        weights = {}
        for i, factor_name in enumerate(icir_scores.keys()):
            w = np.clip(softmax_weights[i], self.config.min_weight, self.config.max_weight)
            weights[factor_name] = float(w)
        
        # 재정규화
        total = sum(weights.values())
        if total > 0:
            weights = {k: v/total for k, v in weights.items()}
        
        return weights
    
    def get_diagnostics(self) -> Dict:
        """진단 정보"""
        result = {}
        for factor_name in self._factor_ics:
            ics = self._factor_ics[factor_name][-self.config.lookback_period:]
            if len(ics) >= 5:
                result[factor_name] = {
                    "mean_ic": np.mean(ics),
                    "std_ic": np.std(ics),
                    "icir": np.mean(ics) / (np.std(ics) + 1e-8),
                    "recent_trend": np.mean(ics[-5:]) - np.mean(ics[-10:-5]) if len(ics) >= 10 else 0,
                }
        return result
```

```python
# tail_risk_hedge_optimizer.py
from dataclasses import dataclass
from typing import Optional, Tuple
import numpy as np

@dataclass
class TRHOConfig:
    """Tail Risk Hedge Optimizer 설정"""
    var_confidence: float = 0.95
    cvar_weight: float = 0.3
    max_hedge_ratio: float = 0.5
    min_hedge_ratio: float = 0.05
    lookback_days: int = 60
    vix_threshold_low: float = 15
    vix_threshold_high: float = 30

class TailRiskHedgeOptimizer:
    """꼬리 위험 기반 헤지 최적화"""
    
    def __init__(self, config: Optional[TRHOConfig] = None):
        self.config = config or TRHOConfig()
        self._returns_history: list[float] = []
    
    def update(self, daily_return: float) -> None:
        """일일 수익률 업데이트"""
        self._returns_history.append(daily_return)
        if len(self._returns_history) > self.config.lookback_days * 2:
            self._returns_history.pop(0)
    
    def calculate_optimal_hedge(self, current_vix: float) -> Tuple[float, dict]:
        """
        최적 헤지 비율 계산
        
        Returns:
            Tuple[float, dict]: (최적 헤지 비율, 진단 정보)
        """
        if len(self._returns_history) < self.config.lookback_days:
            return self.config.min_hedge_ratio, {"status": "insufficient_data"}
        
        returns = np.array(self._returns_history[-self.config.lookback_days:])
        
        # VaR 계산
        var = np.percentile(returns, (1 - self.config.var_confidence) * 100)
        
        # CVaR (Expected Shortfall) 계산
        cvar = returns[returns <= var].mean() if len(returns[returns <= var]) > 0 else var
        
        # 꼬리 위험 점수
        tail_risk_score = abs(cvar) * (1 + self.config.cvar_weight)
        
        # VIX 기반 조정
        if current_vix > self.config.vix_threshold_high:
            vix_multiplier = 1.5
        elif current_vix < self.config.vix_threshold_low:
            vix_multiplier = 0.7
        else:
            vix_multiplier = 1.0
        
        # 최적 헤지 비율
        base_hedge = min(tail_risk_score * 10, 0.3)  # 스케일링
        optimal_hedge = np.clip(
            base_hedge * vix_multiplier,
            self.config.min_hedge_ratio,
            self.config.max_hedge_ratio
        )
        
        diagnostics = {
            "var_95": var,
            "cvar": cvar,
            "tail_risk_score": tail_risk_score,
            "vix_multiplier": vix_multiplier,
            "returns_std": np.std(returns),
            "returns_skew": self._calculate_skewness(returns),
        }
        
        return optimal_hedge, diagnostics
    
    @staticmethod
    def _calculate_skewness(returns: np.ndarray) -> float:
        """왜도 계산"""
        n = len(returns)
        if n < 3:
            return 0.0
        mean = np.mean(returns)
        std = np.std(returns)
        if std < 1e-8:
            return 0.0
        return (n / ((n-1) * (n-2))) * np.sum(((returns - mean) / std) ** 3)


def test_tail_risk_hedge_optimizer():
    """TRHO 단위 테스트"""
    trho = TailRiskHedgeOptimizer()
    
    # 정상 시장
    for _ in range(60):
        trho.update(np.random.normal(0.0005, 0.01))
    
    hedge_ratio, diag = trho.calculate_optimal_hedge(current_vix=18)
    assert trho.config.min_hedge_ratio <= hedge_ratio <= trho.config.max_hedge_ratio
    
    # 위기 시장
    for _ in range(30):
        trho.update(np.random.normal(-0.02, 0.03))
    
    hedge_ratio_crisis, _ = trho.calculate_optimal_hedge(current_vix=35)
    assert hedge_ratio_crisis > hedge_ratio, "위기 시 헤지 비율 증가"
    
    print("✅ TailRiskHedgeOptimizer 테스트 통과")
    return True

if __name__ == "__main__":
    test_tail_risk_hedge_optimizer()
```

```python
# cross_fold_ensemble.py
from dataclasses import dataclass
from typing import Dict, List, Optional
import numpy as np

@dataclass
class CFEConfig:
    """Cross-Fold Ensemble 설정"""
    n_models: int = 3
    min_sharpe_threshold: float = 1.0
    diversity_weight: float = 0.3
    rebalance_frequency: int = 20  # 일

class CrossFoldEnsemble:
    """폴드 간 앙상블 학습"""
    
    def __init__(self, config: Optional[CFEConfig] = None):
        self.config = config or CFEConfig()
        self._model_weights: Dict[str, float] = {}
        self._model_sharpes: Dict[str, List[float]] = {}
        self._correlation_matrix: Optional[np.ndarray] = None
    
    def update_model_performance(self, model_id: str, sharpe: float, returns: List[float]) -> None:
        """모델 성능 업데이트"""
        if model_id not in self._model_sharpes:
            self._model_sharpes[model_id] = []
        self._model_sharpes[model_id].append(sharpe)
    
    def calculate_ensemble_weights(self) -> Dict[str, float]:
        """앙상블 가중치 계산"""
        if len(self._model_sharpes) == 0:
            return {}
        
        # 최근 Sharpe 평균
        avg_sharpes = {}
        for model_id, sharpes in self._model_sharpes.items():
            if len(sharpes) >= 3:
                avg_sharpes[model_id] = np.mean(sharpes[-5:])
        
        if not avg_sharpes:
            return {k: 1/len(self._model_sharpes) for k in self._model_sharpes}
        
        # Sharpe 기반 가중치
        min_sharpe = min(avg_sharpes.values())
        adjusted = {k: max(v - min_sharpe + 0.1, 0.1) for k, v in avg_sharpes.items()}
        total = sum(adjusted.values())
        
        return {k: v/total for k, v in adjusted.items()}
    
    def get_ensemble_signal(self, model_signals: Dict[str, float]) -> float:
        """앙상블 신호 생성"""
        weights = self.calculate_ensemble_weights()
        
        if not weights:
            return np.mean(list(model_signals.values())) if model_signals else 0.0
        
        ensemble_signal = 0.0
        total_weight = 0.0
        
        for model_id, signal in model_signals.items():
            if model_id in weights:
                ensemble_signal += signal * weights[model_id]
                total_weight += weights[model_id]
        
        return ensemble_signal / total_weight if total_weight > 0 else 0.0


def test_cross_fold_ensemble():
    """CFE 단위 테스트"""
    cfe = CrossFoldEnsemble()
    
    # 모델 성능 등록
    cfe.update_model_performance("model_a", 2.5, [0.01] * 20)
    cfe.update_model_performance("model_a", 2.3, [0.01] * 20)
    cfe.update_model_performance("model_a", 2.4, [0.01] * 20)
    
    cfe.update_model_performance("model_b", 1.5, [0.005] * 20)
    cfe.update_model_performance("model_b", 1.3, [0.005] * 20)
    cfe.update_model_performance("model_b", 1.4, [0.005] * 20)
    
    weights = cfe.calculate_ensemble_weights()
    assert weights["model_a"] > weights["model_b"], "높은 Sharpe 모델에 더 높은 가중치"
    
    # 앙상블 신호
    signals = {"model_a": 0.8, "model_b": 0.3}
    ensemble = cfe.get_ensemble_signal(signals)
    assert 0.3 < ensemble < 0.8, "앙상블 신호는 개별 신호 사이"
    
    print("✅ CrossFoldEnsemble 테스트 통과")
    return True

if __name__ == "__main__":
    test_cross_fold_ensemble()
```

```python
# volatility_regime_scaling.py
from dataclasses import dataclass
from typing import Optional
import numpy as np

@dataclass
class VRSConfig:
    """Volatility Regime Scaling 설정"""
    target_volatility: float = 0.10  # 연간 10%
    max_leverage: float = 1.5
    min_leverage: float = 0.3
    lookback_days: int = 20
    ewm_halflife: int = 10

class VolatilityRegimeScaling:
    """변동성 목표 기반 포지션 스케일링"""
    
    def __init__(self, config: Optional[VRSConfig] = None):
        self.config = config or VRSConfig()
        self._returns: list[float] = []
    
    def update(self, daily_return: float) -> None:
        """일일 수익률 업데이트"""
        self._returns.append(daily_return)
        if len(self._returns) > self.config.lookback_days * 3:
            self._returns.pop(0)
    
    def get_position_scale(self) -> float:
        """현재 변동성 기반 포지션 스케일"""
        if len(self._returns) < self.config.lookback_days:
            return 1.0
        
        returns = np.array(self._returns[-self.config.lookback_days:])
        
        # EWM 변동성
        weights = np.exp(-np.log(2) * np.arange(len(returns))[::-1] / self.config.ewm_halflife)
        weights /= weights.sum()
        
        ewm_var = np.sum(weights * (returns - np.mean(returns)) ** 2)
        realized_vol = np.sqrt(ewm_var * 252)  # 연율화
        
        if realized_vol < 1e-8:
            return 1.0
        
        # 목표 변동성 / 실현 변동성
        scale = self.config.target_volatility / realized_vol
        return np.clip(scale, self.config.min_leverage, self.config.max_leverage)
    
    def get_diagnostics(self) -> dict:
        """진단 정보"""
        if len(self._returns) < self.config.lookback_days:
            return {"status": "insufficient_data"}
        
        returns = np.array(self._returns[-self.config.lookback_days:])
        realized_vol = np.std(returns) * np.sqrt(252)
        
        return {
            "realized_vol_annual": realized_vol,
            "target_vol": self.config.target_volatility,
            "current_scale": self.get_position_scale(),
        }


def test_volatility_regime_scaling():
    """VRS 단위 테스트"""
    vrs = VolatilityRegimeScaling()
    
    # 저변동 시장
    for _ in range(25):
        vrs.update(np.random.normal(0.0005, 0.005))
    
    scale_low = vrs.get_position_scale()
    assert scale_low > 1.0, "저변동 시 레버리지 증가"
    
    # 고변동 시장
    for _ in range(25):
        vrs.update(np.random.normal(0, 0.03))
    
    scale_high = vrs.get_position_scale()
    assert scale_high < 1.0, "고변동 시 레버리지 감소"
    
    print("✅ VolatilityRegimeScaling 테스트 통과")
    return True

    test_volatility_regime_scaling()
```

```python
            adjustments["diagnostics"]["vrs"] = {"scale": vol_scale}
            adjustments["active_modules"].append("vrs")

        adjustments["position_scale"] = max(0.1, min(1.5, adjustments["position_scale"]))
        adjustments["hedge_weight"] = max(0.05, min(0.6, adjustments["hedge_weight"]))

        logger.debug(f"Adjustments: {adjustments}")
```
